// Function to send email using Sendinblue
async function sendEmail(to: string, subject: string, body: string) {
  const res = await fetch("https://api.sendinblue.com/v3/smtp/email", {
    method: "POST",
    headers: {
      "api-key": SENDINBLUE_API_KEY,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      sender: { name: SENDER_NAME, email: SENDER_EMAIL },
      to: [{ email: to }],
      subject: subject,
      textContent: body
    })
  });

  if (!res.ok) {
    console.error("Sendinblue error:", await res.text());
  }
}

// Native Supabase Edge Function handler (no external imports needed)
export default async function handler(req: Request) {
  try {
    const nowUnix = Math.floor(Date.now() / 1000);

    // Fetch tasks where reminder_enabled=1 and reminder_unix is within last minute
    const res = await fetch(`${SUPABASE_URL}/rest/v1/tasks?reminder_enabled=eq.1&reminder_unix=gte.${nowUnix-60}&reminder_unix=lte.${nowUnix}`, {
      headers: {
        "apikey": SUPABASE_KEY,
        "Authorization": `Bearer ${SUPABASE_KEY}`
      }
    });

    const tasks = await res.json();

    for (const task of tasks) {
      const subject = `Reminder: ${task.title}`;
      const body = `Task: ${task.title}\nDescription: ${task.description}\nCategory: ${task.category}`;

      await sendEmail(task.user_email, subject, body);
      console.log(`Reminder sent for task: ${task.title}`);
    }

    return new Response(JSON.stringify({ status: "success", sent: tasks.length }), { status: 200 });
  } catch (err) {
    console.error("Error sending reminders:", err);
    return new Response(JSON.stringify({ status: "error", message: err.message }), { status: 500 });
  }
}
